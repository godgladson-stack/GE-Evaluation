function uploadPaper() { let fileInput = document.getElementById('paperInput').files[0]; if (!fileInput) { alert("Please select a file first!"); return; } let reader = new FileReader(); reader.onload = function(e) { localStorage.setItem("uploadedStudentPaper", e.target.result); alert("Paper uploaded and saved successfully!"); }; reader.readAsDataURL(fileInput); } function submitAssignment() { const Examiner = document.getElementById("Examiner").value; const paperType = document.getElementById("paper").value; const uploadedPaperImage = localStorage.getItem("uploadedStudentPaper"); if (!uploadedPaperImage) { alert("Warning: No uploaded paper image found. Please upload a student paper image first!"); } // Assigning uploaded image base64 or paper name to specific teacher const paperData = uploadedPaperImage || paperType; localStorage.setItem("assignedExaminer_" + examiner, examiner); localStorage.setItem("assignedPaper_" + examiner, paperData); // General fallback save localStorage.setItem("assignedExaminer", examiner); localStorage.setItem("assignedPaper", paperData); alert("Paper (" + paperType + ") assigned successfully to " + examiner); }
